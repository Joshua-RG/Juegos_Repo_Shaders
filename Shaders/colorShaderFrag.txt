void main(){
  	vec2 uv = fragmentUV;
    	float glitch = step(0.95, fract(sin(dot(uv, vec2(12.9898,78.233)) + time) * 43758.5453));
    	vec4 tex = texture(myImage, uv + vec2(glitch * 0.02, 0.0));
    	vec3 colorShift = vec3(
        	0.5 + 0.5 * sin(time + fragmentPosition.x * 10.0),
        	0.5 + 0.5 * sin(time + fragmentPosition.y * 10.0 + 2.0),
        	0.5 + 0.5 * sin(time + fragmentPosition.x * 10.0 + 4.0)
    	);
    	color = tex * vec4(colorShift, 1.0);
}
